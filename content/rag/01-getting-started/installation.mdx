# Installation

To get started, install the core dependencies. We use `pdf-parse` for document extraction and `ai` with `@ai-sdk/openai` for LLM operations.

```bash
npm install ai @ai-sdk/openai pdf-parse zod
```

## The Base Utility (`lib/rag-utils.ts`)

Instead of writing the same vector math on every page, create this utility file. Every technique in this library assumes you have these helpers available.

```typescript
import { embed, cosineSimilarity } from 'ai';
import { openai } from '@ai-sdk/openai';

/**
 * Generates a semantic vector for a string.
 */
export async function getEmbedding(value: string) {
  const { embedding } = await embed({
    model: openai.embedding('text-embedding-3-small'),
    value,
  });
  return embedding;
}

/**
 * Finds the most relevant chunks using cosine similarity.
 */
export function findRelevantChunks(
  queryEmbedding: number[], 
  chunks: { text: string; embedding: number[] }[], 
  topK = 3
) {
  return chunks
    .map((chunk) => ({
      ...chunk,
      similarity: cosineSimilarity(queryEmbedding, chunk.embedding),
    }))
    .sort((a, b) => b.similarity - a.similarity)
    .slice(0, topK);
}
```
